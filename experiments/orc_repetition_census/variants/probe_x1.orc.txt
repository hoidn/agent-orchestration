(workflow-lisp
  (:language "0.1")
  (:target-dsl "2.33")
  (defmodule x1)
  (import std/improve :only (Decision Improvement improve))
  (export reviewed-change)

  (defrecord Brief
    (task String)
    (intent String)
    (repo String))

  (defrecord Change
    (summary String)
    (account String))

  (defrecord Revision
    (summary String)
    (account String)
    (replies String))

  (defrecord Draft
    (account String)
    (replies String)
    (round Int)
    (reason String))

  (defrecord Verdict
    (report String)
    (summary String)
    (findings List[String])
    (redo Bool))

  (defunion Outcome
    (READY (report String) (rounds Int))
    (ESCALATED (question String))
    (UNRESOLVED (reason String) (rounds Int)))

  (defprompt implement-prompt
    (:fills (task :text))
    -> Change
    "Implement this change in the repository:\n\n{task}\n\nReport two things. `summary`: one sentence. `account`: what you changed and why, the alternatives you tried or considered and why you discarded them, and what you did not cover.\n\nWhen you finish, write the required JSON result to the file whose path is in the environment variable ORCHESTRATOR_OUTPUT_BUNDLE_PATH; that file is how your result is received.")

  (defprompt review-prompt
    (:fills (repo :text) (intent :text) (account :text) (replies :text))
    -> Decision[Verdict String]
    "Review the uncommitted changes in the git repository at {repo} for code and architecture quality. Read the code; change nothing except your report and result files.\nIntended change: {intent}\nThe author's account of the change, including what was tried and discarded: {account}\nThe author's replies to earlier findings: {replies}\nJudge four things: (1) the root cause is fixed, not the symptom silenced; (2) no existing validation or guarantee is weakened and nothing changes outside the intended scope; (3) the change holds beyond the reported case; (4) it sits in the layer that owns the behaviour and reuses existing mechanisms instead of adding parallel ones.\nAPPROVE if you would merge it as is, and list anything that does not block merging as findings. REVISE only for defects that must be fixed before merging, each naming the input or scenario that goes wrong; drop a finding the author has answered. REVISE with `redo` true and the reason as `summary` if the design itself must change. BLOCKED, with your question as `reason`, if the decision is not yours. Write your review to a report file, then write the required JSON result to the file whose path is in the environment variable ORCHESTRATOR_OUTPUT_BUNDLE_PATH.")

  (defprompt fix-prompt
    (:fills (task :text) (findings :value List[String]))
    -> Revision
    "A reviewer asks for the changes below. For each finding, either fix it, or leave it and reply why it should not be fixed: a finding can be wrong. Replace code rather than adding a second mechanism beside the first.\n\nTask: {task}\n\nFindings:\n{findings}\n\nReport three things. `summary`: one sentence. `account`: the state of the whole change now, what you changed and why, the alternatives you tried or considered and why you discarded them, and what you did not cover. `replies`: one paragraph for each finding you did not fix, or the word none.\n\nWhen you finish, write the required JSON result to the file whose path is in the environment variable ORCHESTRATOR_OUTPUT_BUNDLE_PATH; that file is how your result is received.")

  (defprompt redo-prompt
    (:fills (task :text) (reason :text))
    -> Change
    "The reviewer rejected the approach. Discard the current changes and implement the task differently.\n\nTask: {task}\n\nWhy the approach was rejected: {reason}\n\nReport two things. `summary`: one sentence. `account`: what you changed and why, the alternatives you tried or considered and why you discarded them, and what you did not cover.\n\nWhen you finish, write the required JSON result to the file whose path is in the environment variable ORCHESTRATOR_OUTPUT_BUNDLE_PATH; that file is how your result is received.")

  (defproc review-draft ((draft Draft) (brief Brief)) -> Decision[Verdict String]
    :effects ((uses-provider providers.reviewer))
    (provider-result providers.reviewer
      :prompt (review-prompt
                :repo brief.repo
                :intent brief.intent
                :account draft.account
                :replies draft.replies)
      :model "gpt-6-sol"
      :effort "high"
      :timeout-sec 5400))

  (defproc revise-draft ((draft Draft) (brief Brief) (verdict Verdict)) -> Draft
    :effects ((uses-provider providers.coder))
    :lowering inline
    (if (= draft.round 3)
      (record-update draft :reason verdict.summary)
      (rework draft brief verdict)))

  (defproc rework ((draft Draft) (brief Brief) (verdict Verdict)) -> Draft
    :effects ((uses-provider providers.coder))
    :lowering private-workflow
    (if verdict.redo
      (let* ((redo (provider-result providers.coder
                     :prompt (redo-prompt :task brief.task :reason verdict.summary)
                     :model "claude-sonnet-5-5"
                     :effort "medium"
                     :timeout-sec 5400)))
        (record-update draft :account redo.account :replies "none" :round (+ draft.round 1)))
      (let* ((revision (provider-result providers.coder
                         :prompt (fix-prompt :task brief.task :findings verdict.findings)
                         :model "claude-sonnet-5-5"
                         :effort "medium"
                         :timeout-sec 5400)))
        (record-update draft :account revision.account :replies revision.replies :round (+ draft.round 1)))))

  (defworkflow reviewed-change
    ((task String)
     (intent String)
     (repo String))
    -> Improvement[Draft Verdict String]
    (let* ((change (provider-result providers.coder
                     :prompt (implement-prompt :task task)
                     :model "claude-sonnet-5-5"
                     :effort "medium"
                     :timeout-sec 5400)))
      (improve (record Draft :account change.account :replies "none" :round 1 :reason "")
               (record Brief :task task :intent intent :repo repo)
               (proc-ref review-draft)
               (proc-ref revise-draft)
               3)))
)
